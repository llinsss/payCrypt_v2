USAGE GUIDE
===============================================================================

LOCAL DEVELOPMENT:
  npm run migrate:dev    # Migrations + all seeds (recommended for dev)
  OR
  npm run migrate        # Schema only
  npm run seed:demo      # All seeds

PRODUCTION DEPLOYMENT:
  npm run migrate        # Schema migrations
  npm run seed:prod      # Production-safe seeds only
  npm run verify:no-demo # Verify no demo data present

DOCKER - PRODUCTION:
  docker run \
    -e MIGRATE_ON_START=prod \
    -e SEED_ON_START=prod \
    paycrypt-backend:latest

DOCKER - DEVELOPMENT:
  docker run \
    -e MIGRATE_ON_START=true \
    -e SEED_ON_START=demo \
    paycrypt-backend:latest
